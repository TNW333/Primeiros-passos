        maior = num3;

        maior = num4;
    }

3 — Laços de Repetição

3.5 — Leia 2 notas, calcule a média, considere aprovação com 9,5 e repita para outro aluno quando a resposta for S.

A documentação mantém a ideia de função para calcular cada aluno e adiciona o contador de aprovados.

Código corrigido/completo

var aprovados = 0;
var continuar = "S";

function nota() {

    let nota1 = parseInt(prompt("Digite a primeira nota:"));
    let nota2 = parseInt(prompt("Digite a segunda nota:"));

    let media = (nota1 + nota2) / 2;

    if (media >= 9.5) {
        alert("Você passou! Média: " + media);
        aprovados += 1;
    } else {
        alert("Sua média ficou abaixo de 9.5: " + media);
    }
}

while (continuar == "S") {

    nota();

    continuar = prompt(
        "Calcular a média de outro aluno? (S/N)"
    ).toUpperCase();

    while (continuar != "S" && continuar != "N") {
        continuar = prompt(
            "Digite uma resposta válida [S/N]"
        ).toUpperCase();
    }
}

alert("Quantidade de alunos aprovados: " + aprovados);

Testar

O teste desta página usa um aluno por vez; a lógica completa está no código ao lado.

No programa original, quando a resposta é S, o processo volta para a leitura das duas notas.
